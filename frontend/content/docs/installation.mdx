## 1. Install CLI & Initialize

Run the initialization command using your preferred package manager. This configures the component registry and installs `framer-motion`:

```bash
# pnpm
pnpm dlx morphy@latest init

# npm
npx morphy@latest init

# bun
bunx morphy@latest init

# yarn
yarn dlx morphy@latest init
```

This creates the `@/components/morphy` folder in your project with full source code ownership.

---

## 2. Wrap Your Root Layout

Add `<MorphyProvider>` inside your root layout (`app/layout.tsx`). This synchronizes route changes and coordinates frame exits/entries:

```tsx title="app/layout.tsx"
import { MorphyProvider } from '@/components/morphy';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <MorphyProvider defaultTransition="cube">
          {children}
        </MorphyProvider>
      </body>
    </html>
  );
}
```

---

## 3. Wrap Your Pages

Wrap each route segment in `<Page>` and define its unique transition:

```tsx title="app/dashboard/page.tsx"
import { Page } from '@/components/morphy';

export default function DashboardPage() {
  return (
    <Page
      transition="circular-portal"
      duration={0.7}
      direction="in"
      className="p-8 max-w-5xl mx-auto"
    >
      <h1 className="text-3xl font-bold">Dashboard</h1>
      <p>Seamlessly transitions with a circular iris portal.</p>
    </Page>
  );
}
```

---

## 4. Add Shaders to Your Project

Use the CLI to pull specific shaders directly into your repository without bloating your bundle:

```bash
# Add 3D Cube Transition
npx morphy add cube

# Add Circular Iris Portal
npx morphy add circular-portal

# Add Origami 3D Blossom
npx morphy add origami-unfold

# Add Fluid Ink Spread
npx morphy add ink-spread
```

---

## 5. Dynamic Switching with `useMorphy()`

To switch transitions programmatically at runtime (e.g. In a user preferences modal or navigation drawer), use the `useMorphy` hook:

```tsx title="components/transition-picker.tsx"
'use client';

import { useMorphy } from '@/components/morphy';

export function TransitionPicker() {
  const { currentTransition, setTransition } = useMorphy();

  return (
    <div className="flex gap-2">
      <button onClick={() => setTransition('cube')}>3D Cube</button>
      <button onClick={() => setTransition('circular-portal')}>Portal</button>
      <button onClick={() => setTransition('origami-unfold')}>Origami</button>
    </div>
  );
}
```
