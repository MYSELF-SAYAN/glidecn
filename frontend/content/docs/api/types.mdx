## Configuration Types

### `TransitionConfig`

Configuration object controlling timing, easing curves, and directional flow:

```ts
export interface TransitionConfig {
  /** Duration in seconds (default: 0.4) */
  duration?: number;

  /** Delay before animation starts, in seconds */
  delay?: number;

  /** Easing preset or custom cubic-bezier tuple */
  ease?: EasingPreset;

  /** Direction for directional transitions */
  direction?: TransitionDirection;

  /** Stagger children delay in seconds */
  stagger?: number;

  /** Arbitrary custom parameters passed to transitions */
  custom?: Record<string, unknown>;
}
```

## Definition Schemas

### `TransitionDefinition`

Schema required to register a custom transition with [`TransitionRegistry`](/docs/api/registry):

```ts
export interface TransitionDefinition {
  /** Metadata for docs and UI listings */
  metadata: TransitionMetadata;

  /** Default configuration values */
  defaultConfig: TransitionConfig;

  /** Resolves Framer Motion variants for this transition */
  getVariants: (config: Required<TransitionConfig>) => TransitionVariants;

  /** Optional Framer Motion transition timing override */
  getTransition?: (config: Required<TransitionConfig>) => import('framer-motion').Transition;
}
```

---

### `TransitionMetadata`

```ts
export interface TransitionMetadata {
  /** Machine identifier (e.g. "fade", "slide") */
  name: string;

  /** Human-readable display title */
  displayName: string;

  /** One-line explanation */
  description: string;

  /** Category family */
  category: TransitionCategory;

  /** Documented prop schema */
  props: TransitionPropSchema[];
}
```

---

### `TransitionVariants`

```ts
import type { Variants } from 'framer-motion';

export interface TransitionVariants {
  /** State before the page enters */
  initial: Variants['initial'];

  /** Active / visible state */
  animate: Variants['animate'];

  /** State when the page exits */
  exit: Variants['exit'];
}
```

## Component & Context Types

### `GlideCNContextValue`

State returned by the [`useGlide()`](/docs/api/hooks/use-glide) hook:

```ts
export interface GlideCNContextValue {
  currentTransition: string;
  transitionDefinition: TransitionDefinition | null;
  config: Required<TransitionConfig>;
  animationState: AnimationState;
  reducedMotion: boolean;
  setTransition: (name: string) => void;
  setConfig: (config: TransitionConfig) => void;
}
```
